with open("data.json", "r") as f:

M = len(TimeRequired)
P = len(TimeRequired[0])


### Define the variables

batches = model.addVars(P, vtype=GRB.INTEGER, lb=0, name="batches")
setupFlags = model.addVars(P, vtype=GRB.BINARY, name="setupFlags")


### Define Big-M (upper bound on batches[p])
# A safe BigM: maximum possible batches limited by machine 1 availability
BigM = max(
    Availability[m] / max(1e-6, min(TimeRequired[m][p] for p in range(P)))
    for m in range(M)
)


### Define the constraints

# Machine capacity constraints for machines 1..M-1 (no setup time)
for m in range(1, M):
    model.addConstr(
        quicksum(TimeRequired[m][p] * batches[p] for p in range(P))
        <= Availability[m],
        name=f"Capacity_machine_{m}"
    )

model.addConstr(
    quicksum(TimeRequired[0][p] * batches[p] + SetupTime[p] * setupFlags[p]
             for p in range(P))

    name="Capacity_machine_0"

for p in range(P):
